EXP3_BFS.py

def bfs(graph, start):
    visited = [start]
    queue = [start]

    while queue:
        node = queue.pop(0)
        print(node, end=" ")

        for neighbour in graph[node]:
            if neighbour not in visited:
                visited.append(neighbour)
                queue.append(neighbour)


graph = {
    "A": ["B", "C"],
    "B": ["D", "E"],
    "C": ["F"],
    "D": [],
    "E": ["F"],
    "F": [],
}

print("BFS Traversal:")
bfs(graph, "A")

# Cost / Complexity
# Time Complexity: O(V + E)
# Space Complexity: O(V)

EXP4_ASTAR.py

from heapq import heappush, heappop


def a_star(graph, heuristic, start, goal):
    queue = [(heuristic[start], 0, start, [start])]
    best_cost = {start: 0}

    while queue:
        _, cost, node, path = heappop(queue)

        if cost > best_cost.get(node, float("inf")):
            continue

        if node == goal:
            return path, cost

        for neighbour, edge_cost in graph.get(node, []):
            new_cost = cost + edge_cost

            if new_cost < best_cost.get(neighbour, float("inf")):
                best_cost[neighbour] = new_cost
                priority = new_cost + heuristic[neighbour]

                heappush(
                    queue,
                    (priority, new_cost, neighbour, path + [neighbour])
                )

    return None, None


graph = {
    "A": [("B", 4), ("C", 3)],
    "B": [("D", 5), ("E", 12)],
    "C": [("E", 10)],
    "D": [("F", 5)],
    "E": [("F", 4)],
    "F": []
}

heuristic = {
    "A": 7,
    "B": 6,
    "C": 4,
    "D": 3,
    "E": 2,
    "F": 0
}

path, cost = a_star(graph, heuristic, "A", "F")

if path:
    print("Least-cost path:", " -> ".join(path))
    print("Total cost:", cost)
else:
    print("No path found")

    net_input = (x1 * w1) + (x2 * w2)

    else:

EXP6_PERCEPTRON.py

def train_perceptron(inputs, targets, learning_rate=0.1, epochs=10):

    weights = [0, 0]
    bias = 0

    for _ in range(epochs):

        for x, target in zip(inputs, targets):

            total = (
                x[0] * weights[0]
                + x[1] * weights[1]
                + bias
            )

            output = 1 if total >= 0 else 0

            error = target - output

            weights[0] += learning_rate * error * x[0]
            weights[1] += learning_rate * error * x[1]

            bias += learning_rate * error

    return weights, bias


def predict(x, weights, bias):

    total = (
        x[0] * weights[0]
        + x[1] * weights[1]
        + bias
    )

    return 1 if total >= 0 else 0


inputs = [
    [0, 0],
    [0, 1],
    [1, 0],
    [1, 1]
]

and_targets = [0, 0, 0, 1]
or_targets = [0, 1, 1, 1]


and_weights, and_bias = train_perceptron(
    inputs,
    and_targets
)

or_weights, or_bias = train_perceptron(
    inputs,
    or_targets
)


print("A B | AND OR")
print("------------")

for x in inputs:

    print(
        x[0],
        x[1],
        "| ",
        predict(x, and_weights, and_bias),
        "  ",
        predict(x, or_weights, or_bias)
    )


print("\nAND weights:", and_weights, "Bias:", and_bias)

print("OR weights:", or_weights, "Bias:", or_bias)

EXP7_HEBB.py

def hebb_learning(inputs, targets):

    weights = [0, 0]
    bias = 0

    for x, target in zip(inputs, targets):

        weights[0] += x[0] * target
        weights[1] += x[1] * target

        bias += target

    return weights, bias


def predict(x, weights, bias):

    net = (
        x[0] * weights[0]
        + x[1] * weights[1]
        + bias
    )

    return 1 if net > 0 else 0


inputs = [
    [0, 0],
    [0, 1],
    [1, 0],
    [1, 1]
]

targets = [0, 0, 0, 1]


weights, bias = hebb_learning(inputs, targets)


print("Hebb Learning Rule - AND Gate")

    output = predict(x, weights, bias)

    print(x[0], " ", x[1], " | ", output)


print("\nFinal weights:", weights)

print("Final bias:", bias)

# EXPERIMENT 9: FUZZY LOGIC CONTROLLER
# Aim: To control fan speed based on temperature and humidity.
#
# Installation:
# python -m pip install numpy scikit-fuzzy networkx

import numpy as np
import skfuzzy as fuzzy
from skfuzzy import control as ctrl


# 1. Define input and output variables

temp = ctrl.Antecedent(
    np.arange(0, 101, 1),
    "temp"
)

humidity = ctrl.Antecedent(
    np.arange(0, 101, 1),
    "humidity"
)

speed = ctrl.Consequent(
    np.arange(0, 101, 1),
    "speed"
)


# 2. Define membership functions

temp["cold"] = fuzzy.trimf(
    temp.universe,
    [0, 0, 20]
)

temp["warm"] = fuzzy.trimf(
    temp.universe,
    [0, 20, 40]
)

temp["hot"] = fuzzy.trimf(
    temp.universe,
    [20, 40, 40]
)


humidity["low"] = fuzzy.trimf(
    humidity.universe,
    [0, 0, 50]
)

humidity["medium"] = fuzzy.trimf(
    humidity.universe,
    [0, 50, 100]
)

humidity["high"] = fuzzy.trimf(
    humidity.universe,
    [50, 100, 100]
)


speed["slow"] = fuzzy.trimf(
    speed.universe,
    [0, 0, 50]
)

speed["medium"] = fuzzy.trimf(
    speed.universe,
    [0, 50, 100]
)

speed["high"] = fuzzy.trimf(
    speed.universe,
    [50, 100, 100]
)


# 3. Define fuzzy rules

rule1 = ctrl.Rule(
    temp["cold"] & humidity["low"],
    speed["slow"]
)

rule2 = ctrl.Rule(
    temp["cold"] & humidity["medium"],
    speed["slow"]
)

rule3 = ctrl.Rule(
    temp["cold"] & humidity["high"],
    speed["medium"]
)

rule4 = ctrl.Rule(
    temp["warm"] & humidity["low"],
    speed["medium"]
)

rule5 = ctrl.Rule(
    temp["warm"] & humidity["medium"],
    speed["medium"]
)

rule6 = ctrl.Rule(
    temp["warm"] & humidity["high"],
    speed["high"]
)

rule7 = ctrl.Rule(
    temp["hot"] & humidity["low"],
    speed["medium"]
)

rule8 = ctrl.Rule(
    temp["hot"] & humidity["medium"],
    speed["high"]
)

rule9 = ctrl.Rule(
    temp["hot"] & humidity["high"],
    speed["high"]
)


# 4. Create the fuzzy control system

speed_ctrl = ctrl.ControlSystem([
    rule1, rule2, rule3,
    rule4, rule5, rule6,
    rule7, rule8, rule9
])

fan = ctrl.ControlSystemSimulation(speed_ctrl)


# 5. Set input values

temperature = 29
humidity_value = 77

fan.input["temp"] = temperature

fan.input["humidity"] = humidity_value


# 6. Compute the fuzzy output

fan.compute()


# 7. Display results

print("Fuzzy Logic Fan Controller")
print("--------------------------")

print("Temperature:", temperature, "°C")

print("Humidity:", humidity_value, "%")

print(
    "Fan Speed:",
    round(fan.output["speed"], 2),
    "%"
)

# EXPERIMENT 10: HYBRID NEURO-FUZZY SYSTEM (ANFIS)
#
# Installation:
# python -m pip install numpy

import numpy as np


# Generalized bell membership function

def bell_mf(x, a, b, c):

    return 1.0 / (
        1.0 + np.abs((x - c) / a) ** (2 * b)
    )


class ANFIS:

    def __init__(self, premise, consequent):

        self.premise = premise

        self.consequent = np.array(
            consequent,
            dtype=float
        )


    def forward(self, x, y, verbose=False):

        # Layer 1: Fuzzification

        muA1 = bell_mf(
            x,
            *self.premise["A1"]
        )

        muA2 = bell_mf(
            x,
            *self.premise["A2"]
        )

        muB1 = bell_mf(
            y,
            *self.premise["B1"]
        )

        muB2 = bell_mf(
            y,
            *self.premise["B2"]
        )


        # Layer 2: Rule firing strengths

        w = np.array([
            muA1 * muB1,
            muA2 * muB2
        ])


        # Layer 3: Normalize firing strengths

        w_bar = w / w.sum()


        # Layer 4: Rule consequents

        inputs = np.array([
            x,
            y,
            1.0
        ])

        f = self.consequent @ inputs

        weighted_f = w_bar * f


        # Layer 5: Final output

        output = weighted_f.sum()


        if verbose:

            print("Layer 1 (Membership grades):")

            print(
                f" muA1({x}) = {muA1:.4f}, "
                f"muA2({x}) = {muA2:.4f}"
            )

            print(
                f" muB1({y}) = {muB1:.4f}, "
                f"muB2({y}) = {muB2:.4f}"
            )


            print("Layer 2 (Firing strengths):")

            print(
                f" w1 = {w[0]:.4f}, "
                f"w2 = {w[1]:.5f}"
            )


            print("Layer 3 (Normalized strengths):")

            print(
                f" w1_bar = {w_bar[0]:.4f}, "
                f"w2_bar = {w_bar[1]:.4f}"
            )


            print("Layer 4 (Rule outputs):")

            print(
                f" f1 = {f[0]:.4f}, "
                f"f2 = {f[1]:.4f}"
            )


            print(
                f" w1_bar*f1 = {weighted_f[0]:.4f}, "
                f"w2_bar*f2 = {weighted_f[1]:.4f}"
            )


            print("Layer 5 (Final output):")

            print(f" O = {output:.4f}")


        # Save values for parameter update

        self.cache = {
            "w_bar": w_bar,
            "inputs": inputs
        }

        return output


    def update_consequent(
        self,
        target,
        output,
        lr,
        verbose=False
    ):

        # Calculate error

        error = target - output

        w_bar = self.cache["w_bar"]

        inputs = self.cache["inputs"]


        # Gradient descent update

        gradient = -error * np.outer(
            w_bar,
            inputs
        )

        delta = -lr * gradient

        old = self.consequent.copy()

        self.consequent += delta


        if verbose:

            names = ["p", "q", "r"]

            print(
                f"{'Param':<6}"
                f"{'-dE/dθ':>10}"
                f"{'Δθ':>10}"
                f"{'Old':>10}"
                f"{'New':>10}"
            )


            for i in range(2):

                for j in range(3):

                    print(
                        f"{names[j]}{i+1:<5}"
                        f"{-gradient[i, j]:>10.4f}"
                        f"{delta[i, j]:>10.4f}"
                        f"{old[i, j]:>10.4f}"
                        f"{self.consequent[i, j]:>10.4f}"
                    )


if __name__ == "__main__":

    # Initial membership function parameters

    premise = {

        "A1": (2, 1, 2),
